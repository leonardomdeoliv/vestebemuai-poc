12/**
13 * Publica o evento PedidoCriado no exchange "pedidos" (RF02).
14 *
15 * A publicacao acontece so DEPOIS do commit da transacao do pedido: assim nunca
16 * sai um evento de um pedido que sofreu rollback, e o checkout nao espera o envio
17 * do e-mail. Se o RabbitMQ estiver fora do ar, o pedido continua gravado e a falha
18 * fica no log (o padrao Outbox fica registrado como melhoria futura no ADR-003).
19 */
20@Component
21public class PublicadorPedidoCriado {
22
23 private static final Logger log = LoggerFactory.getLogger(PublicadorPedidoCriado.class);
24
25 private final RabbitTemplate rabbit;
26
27 public PublicadorPedidoCriado(RabbitTemplate rabbit) {
28 this.rabbit = rabbit;
29 }
30
31 @TransactionalEventListener(phase = TransactionPhase.AFTER_COMMIT)
32 public void publicar(PedidoCriadoEvento evento) {
33 try {
34 rabbit.convertAndSend(RabbitConfig.EXCHANGE_PEDIDOS, RabbitConfig.ROTA_PEDIDO_CRIADO, evento, m -> {
35 m.getMessageProperties().setMessageId(evento.eventoId().toString());
36 m.getMessageProperties().setType("PedidoCriado");
37 return m;
38 });
39 log.info("Evento PedidoCriado {} publicado para o pedido {}", evento.eventoId(), evento.pedidoId());
40 } catch (AmqpException e) {
41 log.error("Pedido {} gravado, mas o evento PedidoCriado nao foi publicado: {}", evento.pedidoId(), e.getMessage());
42 }
43 }
