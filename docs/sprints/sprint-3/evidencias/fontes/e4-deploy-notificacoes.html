1#!/usr/bin/env bash
2# S3.4 - Publicar so o servico de notificacoes e medir o tempo de deploy (RF06 / hipotese H3)
3#
4# 1. Acrescenta um aviso no modelo do e-mail (se ainda nao estiver la).
5# 2. Enquanto envia 1 pedido a cada 0,5 s ao monolitico, reconstroi e republica so o
6# servico (docker compose up -d --build --no-deps notificacoes) e mede o tempo.
7# 3. Confere: todos os pedidos responderam 201, o monolitico nao reiniciou e os
8# e-mails feitos durante a troca sairam com o modelo novo.
9#
10# Uso: ./scripts/deploy-notificacoes.sh ["texto do aviso"] [pasta de saida]
11set -euo pipefail
12cd "$(dirname "$0")/.."
13source scripts/lib/comum.sh
14
15AVISO=${1:-Trocas grátis em até 30 dias pelo site.}
16SAIDA=${2:-docs/sprints/sprint-3/evidencias}
17MODELO=notificacoes-service/src/main/resources/templates/email-confirmacao.html
18PREFIXO=deploy$(date +%H%M%S)
19mkdir -p "$SAIDA"
20
21esperar_url "$API/actuator/health"; esperar_url "$NOTIF/actuator/health"
22docker compose start mailpit > /dev/null 2>&1; esperar_url "$MAILPIT/livez" 30
23INICIO_MONOLITO=$(docker inspect -f '{{.State.StartedAt}}' "$(docker compose ps -q monolito)")
24
25echo ">> 1. Mudanca no modelo do e-mail (so no servico de notificacoes)"
26if ! grep -qF "$AVISO" "$MODELO"; then
27 AVISO_SED=$(printf '%s' "$AVISO" | sed 's/[&|\\]/\\&/g')
28 sed -i "s|<!-- avisos: .*-->|&\n<p class=\"aviso\">$AVISO_SED</p>|" "$MODELO"
29fi
30grep -n "aviso\"" "$MODELO" | sed 's/^/ /'
31
32echo ">> 2. Pedidos chegando ao monolitico durante o deploy (1 a cada 0,5 s)"
33: > "$SAIDA/deploy-build.log"
34CSV="$SAIDA/deploy-pedidos.csv"; echo "pedido,http_status" > "$CSV"
35( i=0; while [ ! -f /tmp/fim-deploy ]; do i=$((i+1))
36 c=$(curl -s -o /dev/null -w '%{http_code}' -X POST "$API/pedidos" -H 'Content-Type: application/json' -d "$(pedido_json "$i" "$PREFIXO")")
37 echo "$i,$c" >> "$CSV"; sleep 0.5; done ) &
38CARGA=$!; rm -f /tmp/fim-deploy; sleep 2
39
40echo ">> 3. docker compose up -d --build --no-deps notificacoes"
41t0=$(agora_ms)
42# PRE_BUILD: comando opcional antes do build (ex.: "mvn -q package" quando o Maven roda fora do Docker)
43[ -n "${PRE_BUILD:-}" ] && bash -c "$PRE_BUILD" >> "$SAIDA/deploy-build.log" 2>&1
44docker compose build notificacoes >> "$SAIDA/deploy-build.log" 2>&1
45t1=$(agora_ms)
46docker compose up -d --no-deps notificacoes > /dev/null 2>&1
47for _ in $(seq 1 120); do
48 [ "$(curl -s "$NOTIF/actuator/health" | python3 -c 'import json,sys
49try: print(json.load(sys.stdin)["status"])
50except Exception: print("")')" = UP ] && break; sleep 0.5
51done
52t2=$(agora_ms)
53sleep 3; touch /tmp/fim-deploy; wait $CARGA; rm -f /tmp/fim-deploy
54
55BUILD=$(awk "BEGIN{printf \"%.1f\", ($t1-$t0)/1000}")
56TROCA=$(awk "BEGIN{printf \"%.1f\", ($t2-$t1)/1000}")
57TOTAL=$(awk "BEGIN{printf \"%.1f\", ($t2-$t0)/1000}")
58echo " build da imagem: ${BUILD} s | troca do conteiner ate o health UP: ${TROCA} s | total: ${TOTAL} s"
59
60ENVIADOS=$(awk 'NR>1' "$CSV" | wc -l); OK=$(awk -F, 'NR>1 && $2==201' "$CSV" | wc -l)
61INICIO_DEPOIS=$(docker inspect -f '{{.State.StartedAt}}' "$(docker compose ps -q monolito)")
62echo ">> 4. Monolitico durante o deploy: $OK de $ENVIADOS pedidos com HTTP 201"
